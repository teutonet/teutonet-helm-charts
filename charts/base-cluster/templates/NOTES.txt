{{- if .Values.rbac.accounts }}

===
Use the following commands to create a kubeconfig for an account, this requires
[kubelogin](https://github.com/int128/kubelogin) (`kubectl oidc-login`) to be
installed and needs the issuer url of your OIDC provider as per your hoster;

Your OIDC provider may prefix or suffix the username/group claims (e.g.
"oidc:someone@example.com" instead of "someone@example.com") before the
apiserver sees them, so the "rbac.accounts" name you bind against might need
to include that. If it's not working, ask your kubernetes hoster or check the
--oidc-username-claim/--oidc-username-prefix/--oidc-groups-claim/--oidc-groups-prefix
flags on your apiserver;

Note that "$user" below is only a local label for the generated kubeconfig
context, the actually authenticated identity is whoever completes the
interactive OIDC login in the browser, and its token is cached by
kubelogin per issuer/client, not per "$user";

function generateKubeconfig() {
  local user="${1:?Missing user parameter}"
  local issuer="${2:?Missing issuer parameter}"
  local cluster="$(kubectl config view --raw -o json | jq --arg context $(kubectl config current-context) -r '.contexts[] | select(.name == $context) | .context.cluster')"

  cat - <<EOF
apiVersion: v1
kind: Config
clusters:
- cluster:
    certificate-authority-data: $(kubectl config view --raw -o json | jq --arg cluster $cluster -r '.clusters[] | select(.name == $cluster) | .cluster["certificate-authority-data"]')
    server: $(kubectl config view --raw -o json | jq --arg cluster $cluster -r '.clusters[] | select(.name == $cluster) | .cluster.server')
  name: $cluster
contexts:
- context:
    cluster: $cluster
    user: ${user}@${cluster}
  name: ${user}@${cluster}
current-context: ${user}@${cluster}
users:
- name: ${user}@${cluster}
  user:
    exec:
      apiVersion: client.authentication.k8s.io/v1beta1
      command: timeout
      args:
        - 1m
        - kubectl
        - oidc-login
        - get-token
        - --oidc-issuer-url=${issuer}
        - --token-cache-storage=keyring
        - --oidc-client-id=kubernetes
        - --oidc-extra-scope=email
        - --oidc-extra-scope=groups
        - --oidc-extra-scope=offline_access
        - --open-url-after-authentication=
EOF
}

# example usage:
$ generateKubeconfig {{ .Values.rbac.accounts | keys | first }} https://your-oidc-issuer.example.com
{{- end -}}
